<script>
(function(){
  function initSmartAdviceQuiz(){
document.querySelectorAll(".qd-quiz-wrap").forEach(function(wrap) {
if (wrap._tsaInit) return;
wrap._tsaInit = true;
const options = wrap.querySelectorAll(".qd-option");
const progressBar = wrap.querySelector(".qd-progress-bar");
const percentage = wrap.querySelector(".qd-percentage");
const attemptedCount = wrap.querySelector("#qd-attemptedCount");
const correctCount = wrap.querySelector("#qd-correctCount");
const wrongCount = wrap.querySelector("#qd-wrongCount");
const message = wrap.querySelector("#qd-message");
const resetButton = wrap.querySelector(".qd-reset-btn");
if (options.length === 0) return;
const totalQuestions = new Set(Array.from(options).map(opt => opt.getAttribute("data-question"))).size;
let attempted = 0;
let correct = 0;
options.forEach(opt => {
let newOpt = opt.cloneNode(true);
opt.parentNode.replaceChild(newOpt, opt);
newOpt.addEventListener("click", function() {
const qId = this.getAttribute("data-question");
const isCorrect = this.getAttribute("data-correct") === "true";
const siblingOpts = wrap.querySelectorAll(`.qd-option[data-question="${qId}"]`);
const alreadyAnswered = Array.from(siblingOpts).some(o => o.classList.contains("qd-answered"));
if (alreadyAnswered) return;
attempted++;
siblingOpts.forEach(o => o.classList.add("qd-answered"));
if (isCorrect) {
correct++;
this.classList.add("qd-correct");
} else {
this.classList.add("qd-wrong");
siblingOpts.forEach(o => {
if (o.getAttribute("data-correct") === "true") {
o.classList.add("qd-correct");
}
});
}
const exp = wrap.querySelector(`.qd-exp[data-question="${qId}"]`);
if (exp) exp.style.display = "block";
if (attemptedCount) attemptedCount.textContent = attempted;
if (correctCount) correctCount.textContent = correct;
if (wrongCount) wrongCount.textContent = attempted - correct;
if (progressBar && percentage) {
const pct = (attempted / totalQuestions) * 100;
progressBar.style.width = pct + "%";
percentage.textContent = Math.round(pct) + "%";
}
if (attempted === totalQuestions && message) {
if (correct === totalQuestions) {
message.textContent = "CONGRATULATIONS! YOU GOT ALL CORRECT!";
} else if (correct >= totalQuestions / 2) {
message.textContent = "GREAT JOB! YOU'RE DOING WELL!";
} else {
message.textContent = "KEEP PRACTICING. YOU CAN IMPROVE!";
}
}
});
});
if (resetButton) {
let newReset = resetButton.cloneNode(true);
resetButton.parentNode.replaceChild(newReset, resetButton);
newReset.addEventListener("click", function() {
wrap.querySelectorAll(".qd-option").forEach(o => {
o.classList.remove("qd-correct", "qd-wrong", "qd-answered");
});
wrap.querySelectorAll(".qd-exp").forEach(e => {
e.style.display = "none";
});
attempted = 0;
correct = 0;
if (attemptedCount) attemptedCount.textContent = "0";
if (correctCount) correctCount.textContent = "0";
if (wrongCount) wrongCount.textContent = "0";
if (progressBar) progressBar.style.width = "0%";
if (percentage) percentage.textContent = "0%";
if (message) message.textContent = "PRACTICE REGULARLY!";
});
}
});
  }
  if (document.readyState === "loading") {
document.addEventListener("DOMContentLoaded", initSmartAdviceQuiz);
  } else {
initSmartAdviceQuiz();
  }
})();
</script>
